Informações para Testes

Cenários de Teste:

  • Validar conteúdo que muda automaticamente
  • Testar captura de conteúdo em momento específico
  • Validar timestamp de atualizações
  • Testar pausa e retomada de atualizações
  • Validar que conteúdo é atualizado periodicamente
  • Testar waits para conteúdo específico

Exemplos de Teste Automatizado:


// Cypress - Validar conteúdo dinâmico
cy.get('[data-testid="conteudo-dinamico"]').should('be.visible')
cy.get('[data-testid="timestamp-dinamico"]').should('contain', 'Atualizado em')
cy.get('[data-testid="conteudo-dinamico"]').then(($el) => {
  const content = $el.text()
  cy.log('Conteúdo capturado:', content)
})

// Pausar e retomar atualização
cy.get('[data-testid="btn-pausar-conteudo"]').click()
cy.get('[data-testid="content-status"]').should('contain', 'pausada')
cy.get('[data-testid="btn-retomar-conteudo"]').click()
                

// Playwright - Aguardar conteúdo específico
await page.getByTestId('conteudo-dinamico').waitFor()
const content = await page.getByTestId('conteudo-dinamico').textContent()
expect(content).toContain('Atualizado')

// Pausar e retomar atualização
await page.getByTestId('btn-pausar-conteudo').click()
await expect(page.getByTestId('content-status')).toContainText('pausada')
await page.getByTestId('btn-retomar-conteudo').click()
                

// Selenium (Java) - Validar conteúdo dinâmico
WebElement dynamic = driver.findElement(By.cssSelector("[data-testid='conteudo-dinamico']"));
WebDriverWait wait = new WebDriverWait(driver, Duration.ofSeconds(5));
wait.until(ExpectedConditions.textToBePresentInElement(dynamic, "Atualizado"));

// Pausar e retomar atualização
driver.findElement(By.cssSelector("[data-testid='btn-pausar-conteudo']")).click();
WebElement status = driver.findElement(By.cssSelector("[data-testid='content-status']"));
assert status.getText().contains("pausada");
                

# Capybara (Ruby) - Validar conteúdo dinâmico
expect(page).to have_css('[data-testid="conteudo-dinamico"]', visible: true)
expect(page).to have_content('Atualizado em', wait: 5)

# Pausar e retomar atualização
find('[data-testid="btn-pausar-conteudo"]').click
expect(find('[data-testid="content-status"]')).to have_content('pausada')
find('[data-testid="btn-retomar-conteudo"]').click
