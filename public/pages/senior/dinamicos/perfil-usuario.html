Informações para Testes

Cenários de Teste:

  • Editar campos do perfil
  • Validar que campos são preenchidos corretamente
  • Testar salvamento de alterações
  • Validar mensagem de sucesso após salvar
  • Testar cancelamento de edição
  • Validar validação de campos obrigatórios

Exemplos de Teste Automatizado:


// Cypress - Editar perfil
cy.get('[data-testid="input-nome-perfil"]').clear().type('João Silva')
cy.get('[data-testid="input-email-perfil"]').clear().type('joao@exemplo.com')
cy.get('[data-testid="textarea-bio-perfil"]').clear().type('Nova bio')
cy.get('[data-testid="btn-salvar-perfil"]').click()
cy.get('[data-testid="perfil-result"]').should('contain', 'salvo')

// Validar valores iniciais
cy.get('[data-testid="input-nome-perfil"]').should('have.value', 'Maria QA')
                

// Playwright - Editar perfil
await page.getByTestId('input-nome-perfil').fill('João Silva')
await page.getByTestId('input-email-perfil').fill('joao@exemplo.com')
await page.getByTestId('textarea-bio-perfil').fill('Nova bio')
await page.getByTestId('btn-salvar-perfil').click()
await expect(page.getByTestId('perfil-result')).toContainText('salvo')
                

// Selenium (Java) - Editar perfil
driver.findElement(By.cssSelector("[data-testid='input-nome-perfil']")).clear();
driver.findElement(By.cssSelector("[data-testid='input-nome-perfil']")).sendKeys("João Silva");
driver.findElement(By.cssSelector("[data-testid='input-email-perfil']")).clear();
driver.findElement(By.cssSelector("[data-testid='input-email-perfil']")).sendKeys("joao@exemplo.com");
driver.findElement(By.cssSelector("[data-testid='btn-salvar-perfil']")).click();
WebElement result = driver.findElement(By.cssSelector("[data-testid='perfil-result']"));
assert result.getText().contains("salvo");
                

# Capybara (Ruby) - Editar perfil
fill_in '[data-testid="input-nome-perfil"]', with: 'João Silva'
fill_in '[data-testid="input-email-perfil"]', with: 'joao@exemplo.com'
fill_in '[data-testid="textarea-bio-perfil"]', with: 'Nova bio'
click_button '[data-testid="btn-salvar-perfil"]'
expect(page).to have_content('salvo', within: '[data-testid="perfil-result"]')
