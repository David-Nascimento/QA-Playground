Informações para Testes

Cenários de Teste:

  • Testar detecção de variante A ou B
  • Validar que diferentes variantes aparecem
  • Testar elementos específicos de cada variante
  • Validar comportamento diferente por variante
  • Testar forçar variante específica
  • Validar que variante é persistente na sessão

Exemplos de Teste Automatizado:


// Cypress - Detectar variante
cy.get('[data-testid="ab-variant"]').then(($el) => {
  const variant = $el.attr('data-variant')
  if (variant === 'A') {
    cy.get('[data-testid="btn-variante-a"]').should('be.visible')
  } else {
    cy.get('[data-testid="btn-variante-b"]').should('be.visible')
  }
})
cy.get('[data-testid="ab-variant"]').should('contain', 'Variante')
                

// Playwright - Detectar variante
const variant = await page.getByTestId('ab-variant').getAttribute('data-variant')
if (variant === 'A') {
  await expect(page.getByTestId('btn-variante-a')).toBeVisible()
} else {
  await expect(page.getByTestId('btn-variante-b')).toBeVisible()
}
                

// Selenium (Java) - Detectar variante
WebElement variantEl = driver.findElement(By.cssSelector("[data-testid='ab-variant']"));
String variant = variantEl.getAttribute("data-variant");
if ("A".equals(variant)) {
  WebElement btnA = driver.findElement(By.cssSelector("[data-testid='btn-variante-a']"));
  assert btnA.isDisplayed();
} else {
  WebElement btnB = driver.findElement(By.cssSelector("[data-testid='btn-variante-b']"));
  assert btnB.isDisplayed();
}
                

# Capybara (Ruby) - Detectar variante
variant = find('[data-testid="ab-variant"]')['data-variant']
if variant == 'A'
  expect(page).to have_css('[data-testid="btn-variante-a"]', visible: true)
else
  expect(page).to have_css('[data-testid="btn-variante-b"]', visible: true)
end

    ${variant}=    Get Element Attribute    css:[data-testid="ab-variant"]    data-variant
    Run Keyword If    '${variant}' == 'A'    Element Should Be Visible    css:[data-testid="btn-variante-a"]
    Run Keyword If    '${variant}' == 'B'    Element Should Be Visible    css:[data-testid="btn-variante-b"]

Elementos identificáveis:

  • data-testid="ab-container" - Container da variante
  • data-testid="ab-variant" - Conteúdo da variante
  • data-testid="btn-trocar-variante" - Botão trocar
  • data-testid="btn-forcar-variante-a" - Botão forçar A
  • data-testid="btn-forcar-variante-b" - Botão forçar B
  • data-testid="ab-result" - Resultado da variante
